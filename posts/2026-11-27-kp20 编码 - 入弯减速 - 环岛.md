这一版改进了环岛，但是是不标准的铜线交叉的环岛，没有切线
交叉环岛的特征电感比较明显，直接检测到特征电感强制转弯就可以

```c
/* ==================== 环岛状态机变量 ==================== */
uint8  ring_state = 0;       // 0:未识别 1:强制入环 2:环内保持 3:强制出环
uint16 ring_timer = 0;       // 环岛状态计时器（5ms一次）
uint8  ring_exit_cnt = 0;    // 出环特征消抖计数
```
在进出环岛时会有些电感值抖动，所以进出电感加入了一些强制执行的代码：
入环岛强制转弯
强制转弯后一段时间禁止执行任何强制代码“强制”执行“非强制”的正常寻迹代码
环岛中分为两版思路
第一版不知道去哪了，是检测导入环岛给固定差速，没有陀螺仪cos有陀螺仪
第二版就是这个环岛内正常寻迹（前提你的S弯足够稳）
中间还有两版融合的尝试，不如直接全部正常寻迹
出环岛速度太快惯性大的话强制向直道方向转向
转向后强制直行

```c
 /* ==================== 环岛状态机（原始值判断） ==================== */
    if (ring_state == 0) 
    {
        // 【状态0：检测入环】
      
        if (Avg_Adc[3] > 730 && Avg_Adc[2] > 30 && Avg_Adc[0] < 450) 
        {
            ring_state = 1;
            ring_timer = 0;
            ring_exit_cnt = 0;
        }
    }
    else if (ring_state == 1) 
    {
        // 【状态1：强制入环】
        ring_timer++;
        Servo_ppd_struct.out = 1500; // 强制给大差速右转切进去 (左快右慢)
//        试跑。如果入环切不进去，把状态1里的 Servo_ppd_struct.out = 1500; 改成 1800。
        if (ring_timer > 60) // 60 * 5ms = 300ms，车身已经完全进入环岛
        {
            ring_state = 2;
            ring_timer = 0;
        }
    }
    else if (ring_state == 2) 
    {
        // 【状态2：环内保持】
        ring_timer++;
        
        // 环内保持顺时针趋势：保留一半原PD + 固定的右转差速偏置(800)
//			如果环内车一直往内圈死打，把状态2里的 + 800 改成 + 500；如果环内车往直道上冲（不转圈），把 + 800 改成 + 1000。


        Servo_ppd_struct.out = (Servo_ppd_struct.out * 0.5) + 800; 
        
        // 【出环判断】左右电感恢复对称，且都有信号
        if (abs(Avg_Adc[0] - Avg_Adc[3]) < -60 && Avg_Adc[0] > 400 && Avg_Adc[3] > 500) /* >>> 根据实测修改这里 <<< */
        {
            ring_exit_cnt++;
            if (ring_exit_cnt > 15) // 连续 75ms 检测到直道特征才判定出环
            {
                ring_state = 3;
                ring_timer = 0;
                ring_exit_cnt = 0;
            }
        }
        else 
        {
            ring_exit_cnt = 0; // 没看到直道，计数清零
        }
        
        // 超时保护：2秒 (400*5ms) 还没出去，强制退出，防止卡死
        if (ring_timer > 400) 
        {
            ring_state = 0;
        }
    }
    else if (ring_state == 3) 
    {
        // 【状态3：强制出环】
        ring_timer++;
        Servo_ppd_struct.out = 0; // 出环回正
        
        if (ring_timer > 20) // 直行 100ms 后恢复普通循迹
        {
            ring_state = 0;
        }
    }
    /* ==================================================================== */

    // 根据标志位分配差速
    if (ring_state != 0) 
    {
        // 环岛过程中，使用入弯速度 (600)，避免速度太快冲出赛道
        target_L = ENTER_TURN_SPEED + (int16)Servo_ppd_struct.out;
        target_R = ENTER_TURN_SPEED - (int16)Servo_ppd_struct.out;
    }
    else if(wan_flag == 1)   // 【正常入弯】减速
    {
        target_L = ENTER_TURN_SPEED + (int16)Servo_ppd_struct.out;
        target_R = ENTER_TURN_SPEED - (int16)Servo_ppd_struct.out;
    }
    else if(exit_turn_cnt > 0)  // 【出弯】减速
    {
        exit_turn_cnt--;
        target_L = SLOW_SPEED_L + (int16)Servo_ppd_struct.out;
        target_R = SLOW_SPEED_R - (int16)Servo_ppd_struct.out;
    }
    else    // 【直道】全速跑
    {
        target_L = target_speed_L + (int16)Servo_ppd_struct.out;
        target_R = target_speed_R - (int16)Servo_ppd_struct.out;
    }

    // 限幅：外轮最高 2500，内轮最低 -900 / -800
    if(target_L >  2500) target_L =  2500;
    if(target_R >  2500) target_R =  2500;
    if(target_L <  -900) target_L =  -900;
    if(target_R <  -800) target_R =  -800;
```
