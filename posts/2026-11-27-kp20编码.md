```c
#include "zf_common_headfile.h"
#include "speed_jianche.h"

/* ==================== 编码器接口定义 ====================
 * 接线要求：
 *   左编码器 PULSE -> P3.4   DIR -> P3.5
 *   右编码器 PULSE -> P0.4   DIR -> P5.3
 * 编码器电源与单片机必须共地！
 * ======================================================= */
#define ENCODER_DIR_1           (TIM0_ENCOEDER)          // 左编码器接口
#define ENCODER_DIR_DIR_1       (IO_P35)                 // 左 DIR 引脚
#define ENCODER_DIR_PULSE_1     (TIM0_ENCOEDER_P34)      // 左 PULSE 引脚

#define ENCODER_DIR_2           (TIM3_ENCOEDER)          // 右编码器接口
#define ENCODER_DIR_DIR_2       (IO_P53)                 // 右 DIR 引脚
#define ENCODER_DIR_PULSE_2     (TIM3_ENCOEDER_P04)      // 右 PULSE 引脚

/* ==================== 测速周期配置 ==================== */
#define PIT_SPEED_CH            (TIM1_PIT)                // 用 TIM1 做测速中断
#define SPEED_SAMPLE_MS         (50)                      // 50ms 采样一次

/* ==================== 方向修正 ====================
 * 如果推车前进时某个轮读到负数，就把对应 SIGN 改成 -1
 * ================================================= */
#define LEFT_DIR_SIGN           (-1)
#define RIGHT_DIR_SIGN          ( 1)

/* ==================== 全局速度变量 ==================== */
/* 每 50ms 更新一次，表示这 50ms 内累计的脉冲数 */
int16 encoder_data_dir_1 = 0;   // 左轮
int16 encoder_data_dir_2 = 0;   // 右轮

/* ==================== 内部函数声明 ==================== */
static void speed_encoder_update(void);

/* ------------------------------------------------------------
 * 函数简介   测速模块初始化
 * 说明       初始化两路带方向编码器 + 注册 50ms 周期中断
 * ------------------------------------------------------------ */
void speed_jianche_Init(void)
{
    /* 1. 初始化两路带方向编码器 */
    encoder_dir_init(ENCODER_DIR_1, ENCODER_DIR_DIR_1, ENCODER_DIR_PULSE_1);
    encoder_dir_init(ENCODER_DIR_2, ENCODER_DIR_DIR_2, ENCODER_DIR_PULSE_2);

    /* 2. 注册 TIM1 周期中断回调 */
    tim1_irq_handler = speed_encoder_update;

    /* 3. 启动 50ms 周期中断 */
    pit_ms_init(PIT_SPEED_CH, SPEED_SAMPLE_MS);
}

/* ------------------------------------------------------------
 * 函数简介   50ms 周期中断回调
 * 说明       读编码器计数 -> 保存 -> 清零
 * ------------------------------------------------------------ */
static void speed_encoder_update(void)
{
    encoder_data_dir_1 = LEFT_DIR_SIGN  * encoder_get_count(ENCODER_DIR_1);
    encoder_clear_count(ENCODER_DIR_1);

    encoder_data_dir_2 = RIGHT_DIR_SIGN * encoder_get_count(ENCODER_DIR_2);
    encoder_clear_count(ENCODER_DIR_2);
}

/* ------------------------------------------------------------
 * 函数简介   读取左轮速度
 * 返回       每 50ms 脉冲数，正=正转，负=反转
 * ------------------------------------------------------------ */
int16 speed_left(void)
{
    return encoder_data_dir_1;
}

/* ------------------------------------------------------------
 * 函数简介   读取右轮速度
 * 返回       每 50ms 脉冲数，正=正转，负=反转
 * ------------------------------------------------------------ */
int16 speed_right(void)
{
    return encoder_data_dir_2;
}
```
