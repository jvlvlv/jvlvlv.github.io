实际上十字路口过不好的原因99%是因为实验室地下有电线扰乱了电感值，后来的赛道都没有这种问题

```c
/* ==================== 出弯减速功能 ==================== */
#define  EXIT_TURN_SLOW_TIME   350
#define  SLOW_SPEED_L          500
#define  SLOW_SPEED_R          500

uint16 exit_turn_cnt = 0;
bit    last_wan_flag = 0;
/* ==================================================== */

DoublePD_TypeDef Servo_ppd_struct;

/* ============================================================
 * target_speed_L / target_speed_R（目标速度，50ms 脉冲数）
 *
 * 你的标定数据：
 *   PWM=1000 → 速度约 590
 *   PWM=1500 → 左1130 右1250
 *   PWM=2000 → 左1700 右1800
 *
 * 目标建议先设 1200，两边都能达到
 * ============================================================ */
int target_speed_L = 700;
int target_speed_R = 700;

float k_bu = 1.2;

int out_L, out_R, Sp_R = 0, Sp_L = 0, Last_Out = 0;

bit huan_flag = 1;

/* ============================================================
 * 速度环 PID 对象
 * ============================================================ */
SpeedPID_TypeDef Speed_PID_L;
SpeedPID_TypeDef Speed_PID_R;

/* ============================================================
 * 速度环总开关
 *   0 = 开环
 *   1 = 闭环
 * ============================================================ */
uint8 speed_loop_enable = 1;

/* 最终输出给电机的 PWM（速度环每 50ms 更新一次） */
static int pwm_out_L = 0;
static int pwm_out_R = 0;

/* 8MS 分频计数，5ms × 10 = 50ms */
static uint8 speed_div_cnt = 0;

/* ============================================================ */

extern bit wan_flag;
extern float Adc_LEFT1_Value, Adc_LEFT2_Value, Adc_RIGHT1_Value, Adc_RIGHT2_Value, Avg_Adc[];

void _8MS_Main(void);

void _8ms_Init(void)
{
    pit_ms_init(PIT_CH4, 5);
    tim4_irq_handler = _8MS_Main;

    /* 初始化速度环 PID */
SpeedPID_Init(&Speed_PID_L, 2.0f, 0.20f, 0.0f, 2500.0f, 10000.0f);
SpeedPID_Init(&Speed_PID_R, 2.0f, 0.20f, 0.0f, 2500.0f, 10000.0f);
	//SpeedPID_Init(&Speed_PID_L, 3.0f, 0.30f, 0.0f, 2500.0f, 10000.0f);
//SpeedPID_Init(&Speed_PID_R, 3.0f, 0.30f, 0.0f, 2500.0f, 10000.0f);
}

void _8MS_Main(void)
{
    int16 target_L, target_R;

    /* ================================================================
     * 第一部分：循迹（每 5ms 都跑）
     * ================================================================ */
    Adc_Read();
    Adc_Count();
    Error_Count();
    DoublePD_Calc(&Servo_ppd_struct, error, 0);

    if(wan_flag && Servo_ppd_struct.out > 0) k_bu = 1.35;

    if(Servo_ppd_struct.out < 0) Servo_ppd_struct.out = Servo_ppd_struct.out * 1.35;

    /* 出弯后小直道减速 */
    if(last_wan_flag == 1 && wan_flag == 0)
    {
        exit_turn_cnt = EXIT_TURN_SLOW_TIME;
    }
    last_wan_flag = wan_flag;

    if(exit_turn_cnt > 0)
    {
        exit_turn_cnt--;
        target_L = SLOW_SPEED_L + (int16)Servo_ppd_struct.out;
        target_R = SLOW_SPEED_R - (int16)Servo_ppd_struct.out;
    }
    else
    {
        target_L = target_speed_L + (int16)Servo_ppd_struct.out;
        target_R = target_speed_R - (int16)Servo_ppd_struct.out;
    }

    if(target_L >  2500) target_L =  2500;
    if(target_R >  2900) target_R =  2500;
    if(target_L <  -1400) target_L =  -1400;
    if(target_R <  -1000) target_R =  -1000;

    out_L = target_L;
    out_R = target_R;

    /* ================================================================
     * 第二部分：速度环（每 50ms 跑一次，和 TIM1 采样同步）
     * ================================================================ */
    speed_div_cnt++;
    if(speed_div_cnt >= 10)     /* 5ms × 10 = 50ms */
    {
        speed_div_cnt = 0;

        /* 不调用 speed_jianche_Sample()！
         * TIM1 中断已经在后台每 50ms 自动更新 encoder_data_dir_1/2 */

        if(speed_loop_enable)
        {
            pwm_out_L = (int)SpeedPID_Calc(&Speed_PID_L, (float)target_L, (float)encoder_data_dir_1);
            pwm_out_R = (int)SpeedPID_Calc(&Speed_PID_R, (float)target_R, (float)encoder_data_dir_2);
        }
        else
        {
            pwm_out_L = target_L;
            pwm_out_R = target_R;
        }
    }

    /* ================================================================
     * 第三部分：输出给电机
     * ================================================================ */
    moter_left (pwm_out_L);
    moter_right(pwm_out_R);
}

/* ============================================================
 * 双PD控制器（保持不变）
 * ============================================================ */
void DoublePD_Calc(DoublePD_TypeDef *pd, float current_val, float target_val)
{
    float error;
    float p1, p2, d;
    error = target_val - current_val;
    p1 = error * pd->kp1;
    p2 = fabs(error) * error * pd->kp2;
    d  = (error - pd->last_error) * pd->kd;
    pd->out = p1 + p2 + d;
    pd->last_error = error;
}

void DoublePD_Init(DoublePD_TypeDef *pd, float kp1, float kp2, float kd)
{
    pd->kp1 = kp1;
    pd->kp2 = kp2;
    pd->kd  = kd;
    pd->last_error = 0.0f;
}
```
