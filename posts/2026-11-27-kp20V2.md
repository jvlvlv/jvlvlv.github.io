这一版是无编码器原本坏电机的最后一版，这时候已经能勉强跑完除了环岛坡道以外的全程了
相比于kp20没有太多算法上的更新，只是进行了大量的分段来应对不同的实际情况
在下一版就是更换车体，增加编码器的代码了
```c
// ==================== 速度相关参数 ====================
// 直道正常速度
int  SPEED_STRAIGHT_L  =1250,  SPEED_STRAIGHT_R=   1100;
// 入弯速度（比直道低，防止冲出去）
#define  SPEED_TURN_L           650
#define  SPEED_TURN_R           500
// 出弯减速阶段速度（比入弯略高，平滑过渡）
#define  SLOW_SPEED_L           950
#define  SLOW_SPEED_R           900

// 出弯减速持续时间(单位: 5ms * 600 = 3秒)，按需调整
#define  EXIT_TURN_SLOW_TIME    350
// ======================================================

DoublePD_TypeDef Servo_ppd_struct;

//int target_speed_L=1500,target_speed_R=700;
int target_speed_L=1250,target_speed_R=1100;//1050，800
float k_bu=1.2;

int out_L, out_R,Sp_R=0,Sp_L=0,Last_Out=0;

bit huan_flag=1;

//uint8 Role_flag=0;

extern bit wan_flag;
extern float Adc_LEFT1_Value,Adc_LEFT2_Value,Adc_RIGHT1_Value,Adc_RIGHT2_Value,Avg_Adc[];
void _8MS_Main(void);

/* ==================== 出弯减速功能变量 ==================== */
uint16 exit_turn_cnt   = 0;      // 出弯减速计时器
bit    last_wan_flag   = 0;      // 上一次的弯道标志
/* ======================================================== */

void _8ms_Init(void)
{
    pit_ms_init(PIT_CH4, 5);
    tim4_irq_handler = _8MS_Main;
}

void _8MS_Main(void)
{
//		pid_vaihuan(&out_L, &out_R);原本的pid可惜不好用，留个念想

    /* ==================== 1. 采集电感 + 算误差 ==================== */
    Adc_Read();      // 读取电感值
    Adc_Count();     // 电感值归一化
    Error_Count();   // 计算error值 + 更新wan_flag(带消抖)

    /* ==================== 2. 分段 kp/kd ==================== */
    // 直道柔和，弯道激进；kd 用来压超调，防止"拐过弯"
    if(wan_flag)   // 弯道
    {
        Servo_ppd_struct.kp1 = 23.0f;   // 大 kp，快速转向
        Servo_ppd_struct.kp2 = 0.0f;    // 非线性项，误差越大加得越多
        Servo_ppd_struct.kd  = 0.0f;    // 关键：加阻尼，压超调
    }
//    else           // 直道
//    {
//        Servo_ppd_struct.kp1 = 16.0f;   // 小 kp，稳定不抖
//        Servo_ppd_struct.kp2 = 0.0f;
//        Servo_ppd_struct.kd  = 0.0f;    // 直道 kd 小一点，避免噪声放大
    

    DoublePD_Calc(&Servo_ppd_struct, error, 0);

    /* ==================== 3. 基础目标速度：入弯降速 ==================== */
    if(wan_flag)   // 弯道：立即降速
    {
        target_speed_L = SPEED_TURN_L;
        target_speed_R = SPEED_TURN_R;
    }
    else           // 直道：恢复直道速度
    {
        target_speed_L = SPEED_STRAIGHT_L;
        target_speed_R = SPEED_STRAIGHT_R;
    }

    /* ==================== 4. 出弯后小直道减速 ==================== */
    // 检测下降沿：上一次是弯道(1)，这一次是直道(0) -> 刚从弯道出来
    if(last_wan_flag == 1 && wan_flag == 0)
    {
        exit_turn_cnt = EXIT_TURN_SLOW_TIME;   // 重新装载计时
    }
    last_wan_flag = wan_flag;                  // 更新历史标志

    // 出弯减速阶段：覆盖目标速度为减速值
    if(exit_turn_cnt > 0)
    {
        exit_turn_cnt--;
        target_speed_L = SLOW_SPEED_L;
        target_speed_R = SLOW_SPEED_R;
    }
    /* ====================================================== */

    /* ==================== 5. 计算输出 ==================== */
    out_L = target_speed_L + Servo_ppd_struct.out;
    out_R = target_speed_R - Servo_ppd_struct.out;

    /* ==================== 6. 左转补偿 ==================== */
    // 观察到左转error有点小，进行一下简单的补偿
    if(out_L < 0) out_L = out_L * 1.35;

    /* ==================== 7. 限幅 ==================== */
    if(out_L >  3000) out_L =  3000;
    if(out_R >  3000) out_R =  3000;
    if(out_L < 0) out_L = -900;
    if(out_R < 0) out_R = -800;

    /* ==================== 8. 输出到电机 ==================== */
    moter_left (out_L);
    moter_right(out_R);
}
```
